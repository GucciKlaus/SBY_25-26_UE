malloc() --> Speicherallokations - Funktion

in 'vim':


	#include <stdlib.h>

	int main (int argc, char** argv, char** enpv) {

		char* newmemory = malloc(100);

		if(newmemory) {

			newmemory[32] = 42;

			free(newmemory);	//--> keine groessen - Angabe, da Speicherallokation und groesse von selbst geschieht ohne User Input | mann muss nur den Startwert angeben

			char * anothermemory = malloc(100); --> in Byte

			if(anothermemory) {

				anothermemory[1] = 24;

				for int I = 0; ; i++) {		//--> beschreibt den Speicher endlos (bis zur 'Endaddresse' unseres Heaps (Da kein weiterer Speicher angefragt wird)

					anothermemory[i] =123;

				{

				free(anothermemory);	//--> gibt Speicher wieder zum Beschreiben frei

				newmemory = 0;		//--> keine Laufzeitfehler wenn diese Zeile nicht waere (Trotz der Fehlerhaften naechsten Zeile)

				newmemory[24] = 123;

				return 0;
			}

		}

		return -1;
	}
